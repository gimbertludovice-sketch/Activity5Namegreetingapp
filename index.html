package com.example.namegreetingapp;

import android.content.Intent;
import android.os.Bundle;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;

/**
 * Activity 5 - Name Greeting App
 * Student: Gimbert Ludovice | Section: WITMWD3M1
 * Instructor: Sir John Kyle Martin
 */
public class MainActivity extends AppCompatActivity {

    private EditText etName;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        etName = findViewById(R.id.etName);
        Button btnSubmit = findViewById(R.id.btnSubmit);

        btnSubmit.setOnClickListener(v -> {
            String name = etName.getText().toString().trim();

            if (name.isEmpty()) {
                Toast.makeText(MainActivity.this, "Please enter your name", Toast.LENGTH_SHORT).show();
            } else {
                // Send name to DisplayActivity using Intent Extra
                Intent intent = new Intent(MainActivity.this, DisplayActivity.class);
                intent.putExtra("USER_NAME", name);
                startActivity(intent);
            }
        });
    }
}
package com.example.activity5_namegreetingapp;

import android.os.Bundle;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import android.widget.TextView;
import androidx.appcompat.app.AppCompatActivity;

public class DisplayActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_display);

        TextView tvGreeting = findViewById(R.id.tvGreeting);
        WebView webViewPortfolio = findViewById(R.id.webViewPortfolio);

        // Retrieve the extra name sent from MainActivity
        String userName = getIntent().getStringExtra("USER_NAME");

        if (userName != null && !userName.isEmpty()) {
            tvGreeting.setText("Hello, " + userName + "!");
        } else {
            tvGreeting.setText("Hello, Guest!");
        }

        // Configure WebView to render responsive HTML5 Portfolio
        WebSettings webSettings = webViewPortfolio.getSettings();
        webSettings.setJavaScriptEnabled(true);
        webSettings.setDomStorageEnabled(true);
        webViewPortfolio.setWebViewClient(new WebViewClient());

        // Load direct embedded HTML5 portfolio content
        webViewPortfolio.loadDataWithBaseURL(null, getPortfolioHtml(), "text/html", "UTF-8", null);
    }
}

Prepared by: Gimbert Ludovice (Section WITMWD3M1) | Instructor: Sir John Kyle Martin

1. Android Studio HTML5 WebView Architecture

In Android Studio, embedding HTML5 code inside a native activity requires initializing a WebView view in layout XML and enabling execution via Java/Kotlin code.

  • WebSettings configuration: webSettings.setJavaScriptEnabled(true) allows script execution inside the app frame.
  • Inline Data Loading: Using webView.loadDataWithBaseURL() renders HTML5 markup straight from code strings without requiring extra server setups.

2. Intent Extra Data Passing

When the user inputs their name in MainActivity, the name string is passed seamlessly to DisplayActivity:

Intent intent = new Intent(MainActivity.this, DisplayActivity.class);
intent.putExtra("USER_NAME", name);
startActivity(intent);
